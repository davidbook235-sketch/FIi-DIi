name: FII-DII Scan
on:
  schedule:
    - cron: "30 13 * * 1-5"   # 7:00 PM IST (data aane ke baad)
    - cron: "15 3 * * 1-5"    # 8:45 AM IST (subah ka plan)
  workflow_dispatch:
permissions:
  contents: write
jobs:
  scan:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
      - run: pip install -r requirements.txt
      - name: Run scan
        env:
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
        run: python run_scan.py
      - name: Save data
        run: |
          git config user.name "scanner-bot"
          git config user.email "bot@users.noreply.github.com"
          git add data
          git diff --cached --quiet || git commit -m "scan data update"
          git push
